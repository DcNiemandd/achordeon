# Write your first song

From an empty song to a readable chord sheet. In the app you start in
<AppLink to="songs">`Songs`</AppLink>:
create a song and give it a name. Everything below is the markup you then type —
work through it here, and the render on the right is the app's own.

## Build the song

<SongChallenge
  content={``}
  steps={[
    {
      prompt: 'Write the title: a line that starts with a single asterisk.',
      check: (ast) => !!ast.title,
      solution: `
* Anesthetize
`,
      feedback: [
        {
          when: (ast) => !ast.title,
          message: 'No title yet — start a line with a single asterisk and a space.',
        },
      ],
    },
    {
      prompt: 'Add the subtitle below it: a line that starts with two asterisks.',
      check: (ast) => !!ast.subtitle,
      solution: `
* Anesthetize
** Porcupine Tree
`,
      feedback: [
        {
          when: (ast) => !ast.subtitle,
          message: 'No subtitle yet — start a line with two asterisks and a space.',
        },
      ],
    },
    {
      prompt: 'Type the lyrics: two lines under the subtitle.',
      check: (ast) => ast.blocks.some((b) => b.lines.filter((l) => l.text.trim()).length >= 2),
      solution: `
* Anesthetize
** Porcupine Tree

Wake me up inside
Call my name
`,
      feedback: [
        {
          when: (ast) => ast.blocks.every((b) => b.lines.every((l) => !l.text.trim())),
          message: 'No lyrics yet — write a line under the subtitle.',
        },
        {
          when: (ast) => ast.blocks.every((b) => b.lines.filter((l) => l.text.trim()).length < 2),
          message: 'One line so far — add a second below it.',
        },
      ],
    },
    {
      prompt: 'Label this block Chorus.',
      check: (ast) => ast.blocks[0]?.label === 'Chorus',
      solution: `
* Anesthetize
** Porcupine Tree

Chorus: Wake me up inside
Call my name
`,
      feedback: [
        {
          when: (ast) => !ast.blocks[0]?.label,
          message: 'No label yet — put a word and a colon before the first line.',
        },
        {
          when: (ast) => ast.blocks[0]?.label !== 'Chorus',
          message: 'There is a label, but it is not “Chorus”.',
        },
      ],
    },
    {
      prompt: 'Give the second line its own sub-label: Softly.',
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => l.label === 'Softly')),
      solution: `
* Anesthetize
** Porcupine Tree

Chorus: Wake me up inside
Softly: Call my name
`,
      feedback: [
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => !!l.label)),
          message: 'No sub-label yet — a label on a later line names just that line.',
        },
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => l.label === 'Softly')),
          message: 'There is a sub-label, but it is not “Softly”.',
        },
      ],
    },
    {
      prompt: 'Add another block below — a blank line, then a new line of lyrics.',
      check: (ast) => ast.blocks.filter((b) => b.lines.some((l) => l.text.trim())).length >= 2,
      solution: `
* Anesthetize
** Porcupine Tree

Chorus: Wake me up inside
Softly: Call my name

Save me from the dark
`,
      feedback: [
        {
          when: (ast) => ast.blocks.filter((b) => b.lines.some((l) => l.text.trim())).length < 2,
          message: 'Still one block — a blank line starts a new one.',
        },
      ],
    },
    {
      prompt: 'Here is a fuller song to work on. Put a chord above a word — a real, transposable chord like Am, in brackets right before the letter it lands on.',
      content: `
* Anesthetize
** Porcupine Tree

Chorus: Wake me up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the song begins

Bridge: Bring me to life
hold me while the world goes quiet
`,
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => l.chords.some((c) => c.valid))),
      solution: `
* Anesthetize
** Porcupine Tree

Chorus: Wake me [Am]up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the song begins

Bridge: Bring me to life
hold me while the world goes quiet
`,
      feedback: [
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => l.chords.length > 0)),
          message: 'No chord yet — wrap it in square brackets.',
        },
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => l.chords.some((c) => c.valid))),
          message: 'There is a bracket, but it is not a valid chord — try Am or G.',
        },
      ],
    },
    {
      prompt: 'Add a chords-only line at the top — a row of chords with no words, like an intro.',
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => !l.text.trim() && l.chords.length > 0)),
      solution: `
* Anesthetize
** Porcupine Tree

[Am F G]

Chorus: Wake me [Am]up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the song begins

Bridge: Bring me to life
hold me while the world goes quiet
`,
      feedback: [
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => !l.text.trim() && l.chords.length > 0)),
          message: 'No chords-only line yet — put chords on a line by themselves.',
        },
      ],
    },
    {
      prompt: 'Make one chord read inside the line, with doubled brackets.',
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => l.chords.some((c) => c.inline))),
      solution: `
* Anesthetize
** Porcupine Tree

[Am F G]

Chorus: Wake me [Am]up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the [[G]]song begins

Bridge: Bring me to life
hold me while the world goes quiet
`,
      feedback: [
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => l.chords.some((c) => c.inline))),
          message: 'Single brackets sit above the line — double them to bring the chord inline.',
        },
      ],
    },
    {
      prompt: 'Make a word bold: wrap it in double asterisks, like **this**.',
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => l.spans?.some((s) => s.bold))),
      solution: `
* Anesthetize
** Porcupine Tree

[Am F G]

Chorus: Wake me [Am]up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the [[G]]song begins

Bridge: Bring me to life
hold me while the **world** goes quiet
`,
      feedback: [
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => l.spans?.some((s) => s.bold))),
          message: 'No bold yet — wrap a word in double asterisks: **like this**.',
        },
      ],
    },
    {
      prompt: 'Make a word italic: wrap it in single asterisks, like *this*.',
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => l.spans?.some((s) => s.italic))),
      solution: `
* Anesthetize
** Porcupine Tree

[Am F G]

Chorus: Wake me [Am]up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the [[G]]song begins

Bridge: Bring me to life
hold me while the **world** goes *quiet*
`,
      feedback: [
        {
          when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => l.spans?.some((s) => s.italic))),
          message: 'No italic yet — wrap a word in single asterisks: *like this*.',
        },
      ],
    },
    {
      prompt: 'Add a line that would start with a label, like “Narrator:”, but keep the colon in the words — escape it with a backslash so it stays a lyric.',
      check: (ast) => ast.blocks.some((b) => b.lines.some((l) => !l.label && findLabelDelimiter(l.text) >= 0)),
      solution: `
* Anesthetize
** Porcupine Tree

[Am F G]

Chorus: Wake me [Am]up inside
Softly: Call my name
Save me from the dark

Verse: I'm lying here in silence
before the [[G]]song begins

Bridge: Bring me to life
hold me while the **world** goes *quiet*
Narrator\\: hold on
`,
feedback: [
{
when: (ast) => !ast.blocks.some((b) => b.lines.some((l) => !l.label && findLabelDelimiter(l.text) >= 0)),
message: 'No escaped colon yet — a colon that would start a label needs a backslash before it (an incidental one like 12:30 does not).',
},
],
},
]}
